09 Ripple Click — Material-style ink ripple spawned at click position
▶ live preview — click to trigger
HTML
<button class="ripple-btn"
  onclick="addRipple(event,this)">
  Ripple Click
</button>

<!-- JS: -->
<script>
function addRipple(e, btn) {
  const r = btn.getBoundingClientRect();
  const s = Math.max(r.width, r.height);
  const x = e.clientX - r.left - s/2;
  const y = e.clientY - r.top - s/2;
  const el = document.createElement('span');
  el.className = 'ripple';
  el.style.cssText =
    `width:${s}px;height:${s}px;left:${x}px;top:${y}px`;
  btn.appendChild(el);
  setTimeout(() => el.remove(), 700);
}
</script>
CSS
.ripple-btn {
  position: relative;
  overflow: hidden;
  background: #0ea5e9;
  color: white;
  border: none;
  padding: 14px 36px;
  border-radius: 8px;
  cursor: pointer;
}
.ripple-btn .ripple {
  position: absolute;
  border-radius: 50%;
  background: rgba(255,255,255,0.35);
  transform: scale(0);
  animation: ripple-anim 0.6s linear;
  pointer-events: none;
}
@keyframes ripple-anim {
  to { transform: scale(4); opacity: 0; }
}
